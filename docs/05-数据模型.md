# 数据模型

Postgres 16 + pgvector + RLS。所有业务表带 `workspace_id`，连接级 `SET app.workspace_id`。

## 1. 迁移顺序

```
001  workspaces / workspace_members / role_configs / permission_callbacks / credits_ledger
002  flows / flow_versions / flow_env_vars / run_logs
003  knowledge_bases / kb_files / kb_chunks(+hnsw) / kb_synonyms / kb_versions / kb_operation_logs
004  user_databases / user_tables（+ 独立 data schema，表名前缀 betteryeah*）
005  agents / agent_skills / agent_knowledge / agent_tasks
006  conversations / messages
007  plugins / custom_plugins / mcp_services / skill_packs / a2a_agents
008  eval_datasets / evaluators / eval_runs / monitor_rules / alert_rules
```

## 2. 工作空间与成员

```sql
create table workspaces (
  id uuid primary key,
  name text not null,
  access_key text not null unique,              -- Workspace 级，重新生成即旧值失效
  plan text,
  credits_balance bigint default 0,             -- 积分
  strict_permission_mode boolean default false, -- 复刻开关：off = 与 BetterYeah 一致(fail-open)
  expire_at timestamptz,
  created_at timestamptz default now()
);

create table workspace_members (
  workspace_id uuid not null references workspaces,
  user_id uuid not null,
  role text not null check (role in ('admin','developer','user','viewer')),  -- viewer = 仅查看
  role_config_id uuid,
  primary key (workspace_id, user_id)
);

create table role_configs (                     -- 2025-04-17 角色配置
  id uuid primary key, workspace_id uuid not null,
  name text not null, permissions jsonb not null
);

create table credits_ledger (
  id bigserial primary key, workspace_id uuid not null,
  target_type text, target_id uuid, run_id uuid,
  kind text,                                    -- text | multimodal | knowledge_parse | free
  amount bigint not null, balance_after bigint,
  detail jsonb, at timestamptz default now()
);
```

## 3. Agent

```sql
create table agents (
  id uuid primary key, workspace_id uuid not null references workspaces,
  name text, icon text,
  model_config jsonb,           -- 自由模式/多模态/深度思考/思考过程/增强感知三项
  role_setting jsonb,           -- {mode:'structured'|'text', themes:[{key,weight,content}]}
  rules jsonb,                  -- 任务开关、隐藏技能图示、隐藏技能名称
  variables jsonb,              -- [{name,label,required}]
  mock_variables jsonb,
  permission_callback_id uuid references permission_callbacks,  -- Agent 级差异化
  quick_entries jsonb,          -- 插件/工作流/网页指令
  recommend_questions jsonb,    -- {source:'answer'|'knowledge'|'context', rules}
  tags text[], deleted_at timestamptz
);

create table agent_skills (
  id uuid primary key, agent_id uuid not null references agents,
  skill_type text not null check (skill_type in ('plugin','flow','sub_agent','skill_pack')),
  target_id uuid not null,
  manual text,                  -- 技能说明书
  enabled boolean default true,
  weight int,                   -- 子Agent 权重，越高越优先
  comm_mode text check (comm_mode in ('sync_history','summarize_history','question_only')),
  output_mode text,             -- 输入展示：表单形式等
  force_call boolean default false,
  default_params jsonb, sort int
);

create table agent_knowledge (
  agent_id uuid not null references agents, kb_id uuid not null,
  query_timing text not null check (query_timing in ('force','on_demand')),
  manual text,                  -- 知识库说明书（按需查询必填）
  query_methods text[] not null,-- semantic | keyword | fulltext
  max_results int default 3, rerank boolean default false,
  show_source_link boolean default false,
  primary key (agent_id, kb_id)
);

create table agent_tasks (      -- 异步任务 + 定时重复
  id uuid primary key, agent_id uuid not null references agents,
  skill_id uuid not null references agent_skills,
  cron text, params jsonb,
  preset boolean default false,        -- 预设任务：使用者默认关闭且不可修改
  enabled_by_user boolean default false,
  next_run_at timestamptz, last_run_at timestamptz
);
```

## 4. 工作流

```sql
create table flows (
  id uuid primary key, workspace_id uuid not null references workspaces,
  name text not null, description text,
  multi_env boolean default false,        -- 激活后不可还原
  tags text[], deleted_at timestamptz
);

create table flow_versions (              -- 每次发布一版，可回退
  flow_id uuid not null references flows,
  version int not null,
  env text not null check (env in ('single','dev','test','prod')),
  graph jsonb not null,                   -- 节点 + 边 + 变量声明
  published_at timestamptz, published_by uuid,
  primary key (flow_id, version, env)
);

create table flow_env_vars (              -- 跨环境发布不覆盖
  flow_id uuid not null references flows,
  env text not null, key text not null, value jsonb not null,
  resource_type text,                     -- kb_id | flow_id | model | plain
  primary key (flow_id, env, key)
);

create table flow_schedules (             -- 定时运行；参数创建时固定
  id uuid primary key, flow_id uuid not null references flows,
  env text, interval_spec jsonb not null, params jsonb not null,
  enabled boolean default true, next_run_at timestamptz, last_run_at timestamptz
);

create table run_logs (                   -- Agent 与工作流共用
  id uuid primary key, workspace_id uuid not null,
  target_type text not null check (target_type in ('agent','flow')),
  target_id uuid not null, version int, env text,
  status text not null, nodes jsonb,      -- 每节点入参/出参/耗时
  duration_ms int, first_token_ms int, output_ms int,
  credits bigint, favorited boolean default false,
  trace_id text, at timestamptz default now()
);

create table batch_debug_sets (           -- 批量调试测试集，≤200 条
  id uuid primary key, workspace_id uuid, target_type text, target_id uuid,
  items jsonb not null, concurrency int default 1
);
```

## 5. 知识库

```sql
create table knowledge_bases (
  id uuid primary key, workspace_id uuid not null references workspaces,
  kb_type text not null check (kb_type in ('document','qa','multimodal')),
  name text not null,
  embedding_model text not null,          -- 多语种|纯中文|text-embedding-3-large|jina-v3|jina-clip-v2
  dormant boolean default false,          -- 有效期休眠
  last_used_at timestamptz,               -- 5 个月未使用触发过期
  tags text[], deleted_at timestamptz
);

create table kb_files (
  id uuid primary key, kb_id uuid not null references knowledge_bases,
  name text not null, mimetype text,
  parse_mode text check (parse_mode in ('general','visual_layout')),
  segment_config jsonb,                   -- 分段标识符 / 每段最长字数 / 问题开始字符
  tags text[] check (coalesce(array_length(tags,1),0) <= 9),
  meta_attrs jsonb,                       -- 元属性：自动提取 + 自定义
  source text check (source in ('local','manual','web','video','plugin','feishu','dingtalk','yuque')),
  status text not null,
  valid_from timestamptz, valid_to timestamptz,   -- 问答知识有效期
  parse_ppt_image boolean default false
);

create table kb_chunks (
  id uuid primary key, kb_id uuid not null, file_id uuid not null references kb_files,
  chunk_id int not null, vector_id text,
  content text not null, keywords text[],
  embedding vector(1024),
  embed_status text default 'pending', embed_error text,   -- 单段失败仅标记
  question text, answers jsonb, similar_questions text[],  -- 问答库：1 问 ≤5 答
  media_url text, media_type text, ocr_text text, ai_caption text  -- 多模态
);
create index on kb_chunks using hnsw (embedding vector_cosine_ops);
create index on kb_chunks using gin (keywords);

create table kb_synonyms (              -- 实体词 1 个≤20字；近义词 ≤25 个且各≤20字
  kb_id uuid not null references knowledge_bases,
  entity text not null, synonyms text[] not null, primary key (kb_id, entity)
);
create table kb_versions (file_id uuid not null, version int not null,
  snapshot jsonb not null, created_at timestamptz default now(), primary key (file_id, version));
create table kb_operation_logs (kb_id uuid not null, action text not null,
  operator uuid, detail jsonb, at timestamptz default now());
```

## 7. 会话与消息

```sql
create table conversations (
  id uuid primary key, agent_id uuid not null references agents,
  user_type text, user_id text,           -- 渠道身份，见 06-API契约 权限 SPI
  title text, client_type text,           -- PC | MOBILE | DINGTALK_WEB | CLIENT
  variables jsonb,                        -- 会话前填写的 Agent 变量
  meta_data jsonb,                        -- SDK agentMetadata → betterAI.session.metaData
  created_at timestamptz default now()
);

create table messages (
  id uuid primary key, conversation_id uuid not null references conversations,
  role text not null check (role in ('user','assistant','system')),
  content text,
  tasks jsonb,                            -- name/type/tool_type/tool_id/status/content/metadata
  first_token_ms int, output_ms int, duration_ms int,
  credits bigint, feedback text,           -- 不喜欢/点踩
  at timestamptz default now()
);

create table session_store (              -- betterAI.session.addChatRecord / initSessionData
  conversation_id uuid not null references conversations,
  key text not null, value jsonb, primary key (conversation_id, key)
);
```

## 8. 插件 / MCP / Skill / A2A

```sql
create table plugins (                    -- 官方插件（内置，也称系统插件）
  id uuid primary key, code text not null unique, name text not null,
  description text,                       -- 进 prompt，影响 AI 调用效果
  input_schema jsonb not null, output_schema jsonb,
  category text, credits_cost bigint default 0   -- 图文分类器等为 0
);

create table custom_plugins (
  id uuid primary key, workspace_id uuid not null references workspaces,
  name text not null, description text,
  create_mode text not null check (create_mode in ('api_doc','manual','mcp_custom','mcp_sync')),
  auth_type text check (auth_type in ('static_token','auth_flow')),
  auth_config jsonb,                      -- static: token; auth_flow: {flowId, inputs, ttl}
  auth_expire_at timestamptz              -- 达到失效条件后平台自动重新鉴权
);
create table custom_plugin_tools (
  id uuid primary key, plugin_id uuid not null references custom_plugins,
  name text not null, description text not null,   -- 必须让 AI 理解怎么用
  code text, input_schema jsonb not null
);

create table mcp_services (
  id uuid primary key, workspace_id uuid not null,
  service_id text not null,               -- AI 依 ID 调用，命名需可解释用途
  name text not null, description text, published boolean default false,
  unique (workspace_id, service_id)
);
create table mcp_service_tools (
  mcp_id uuid not null references mcp_services,
  target_type text not null check (target_type in ('flow','plugin','custom_plugin','kb')),
  target_id uuid not null, tool_desc text, input_desc jsonb,
  primary key (mcp_id, target_type, target_id)
);

create table skill_packs (                -- 2026-03-05
  id uuid primary key, workspace_id uuid not null,
  name text not null, description text not null,     -- Skill 描述
  tools jsonb not null,                   -- 工作流 + 插件 + 内部工具用法说明
  published boolean default false
);

create table a2a_agents (
  id uuid primary key, workspace_id uuid not null,
  agent_id uuid references agents,        -- 自建 Agent 发布为 A2A
  external_endpoint text,                 -- 或连接外部 A2A
  name text, description text
);
```

## 9. 评测与监控

```sql
create table eval_datasets (
  id uuid primary key, workspace_id uuid not null, name text not null, items jsonb not null
);
create table evaluators (                 -- 用工作流编排评估逻辑
  id uuid primary key, workspace_id uuid not null, name text not null,
  flow_id uuid not null references flows,
  dimensions jsonb                        -- 自定义评估维度
);
create table eval_runs (
  id uuid primary key, workspace_id uuid not null,
  target_type text not null check (target_type in ('agent','flow')), target_id uuid not null,
  dataset_id uuid not null references eval_datasets,
  evaluator_id uuid not null references evaluators,
  results jsonb,                          -- 每条 {score, analysis, overall_evaluation}
  aggregate jsonb, created_at timestamptz default now()
);

create table monitor_rules (              -- 效果监控
  id uuid primary key, workspace_id uuid not null, name text not null,
  target_type text, target_id uuid,
  rule jsonb not null, chart jsonb, template text,
  advanced_flow_id uuid references flows  -- 高级编排模式
);

create table alert_rules (
  id uuid primary key, workspace_id uuid not null,
  event text not null check (event in
    ('credits_low','credits_fast','app_error','timeout','workspace_expire')),
  channels jsonb not null,                -- sms | dingtalk_group(@member) | phone
  threshold jsonb, enabled boolean default true
);
```
